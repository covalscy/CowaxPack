    // Методы для получения позиций костей стволов
    public Vec3 getLeftBarrel1Position(float ticks) {
        Matrix4f transformT = getBarrelTransformFromTurret(ticks);
        
        Matrix4f transform = new Matrix4f();
        // Координаты левого ствола относительно башни
        Vector4f worldPosition = transformPosition(transform, 0f, 0f, 0f);

        transformT.translate(worldPosition.x, worldPosition.y, worldPosition.z);

        float a = getTurretYaw(ticks);

        float r = (Mth.abs(a) - 90f) / 90f;

        float r2;

        if (Mth.abs(a) <= 90f) {
            r2 = a / 90f;
        } else {
            if (a < 0) {
                r2 = -(180f + a) / 90f;
            } else {
                r2 = (180f - a) / 90f;
            }
        }

        float x = Mth.lerp(ticks, turretXRotO, getTurretXRot());
        float xV = Mth.lerp(ticks, xRotO, getXRot());
        float z = Mth.lerp(ticks, prevRoll, getRoll());

        transformT.rotate(Axis.XP.rotationDegrees(x + r * xV + r2 * z));
        
        return transformT;
    }

        return finalPosition;
    }

    @Override
    public void baseTick() {
        super.baseTick();
        turretYRotO = this.getTurretYRot();
        this.updateOBB();

        if (this.entityData.get(GUN_FIRE_TIME) > 0) {
            this.entityData.set(GUN_FIRE_TIME, this.entityData.get(GUN_FIRE_TIME) - 1);
        }

        if (this.level() instanceof ServerLevel) {
            this.handleAmmo();
        }

        if (this.onGround()) {
            float f0 = 0.6f + 0.25f * Mth.abs(90 - (float) calculateAngle(this.getDeltaMovement(), this.getViewVector(1))) / 90;
            this.setDeltaMovement(this.getDeltaMovement().add(this.getViewVector(1).normalize().scale(0.05 * getDeltaMovement().dot(getViewVector(1)))));
            this.setDeltaMovement(this.getDeltaMovement().multiply(f0, 0.99, f0));
        } else {
            this.setDeltaMovement(this.getDeltaMovement().multiply(0.99, 0.99, 0.99));
        }

        if (this.isInWater()) {
            float f1 = (float) (0.7f - (0.04f * Math.min(getSubmergedHeight(this), this.getBbHeight())) + 0.08f * Mth.abs(90 - (float) calculateAngle(this.getDeltaMovement(), this.getViewVector(1))) / 90);
            this.setDeltaMovement(this.getDeltaMovement().add(this.getViewVector(1).normalize().scale(0.04 * getDeltaMovement().dot(getViewVector(1)))));
            this.setDeltaMovement(this.getDeltaMovement().multiply(f1, 0.85, f1));
        }

        if (this.level() instanceof ServerLevel serverLevel && this.isInWater() && this.getDeltaMovement().length() > 0.1) {
            sendParticle(serverLevel, ParticleTypes.CLOUD, this.getX() + 0.5 * this.getDeltaMovement().x, this.getY() + getSubmergedHeight(this) - 0.2, this.getZ() + 0.5 * this.getDeltaMovement().z, (int) (2 + 4 * this.getDeltaMovement().length()), 0.65, 0, 0.65, 0, true);
            sendParticle(serverLevel, ParticleTypes.BUBBLE_COLUMN_UP, this.getX() + 0.5 * this.getDeltaMovement().x, this.getY() + getSubmergedHeight(this) - 0.2, this.getZ() + 0.5 * this.getDeltaMovement().z, (int) (2 + 10 * this.getDeltaMovement().length()), 0.65, 0, 0.65, 0, true);
        }

        turretAngle(10, 12.5f);
        gunnerAngle(15, 15);
        lowHealthWarning();
        

        terrainCompact(4.0f, 5.0f);

        inertiaRotate(1.2f);

        // releaseSmokeDecoy(getTurretVector(1)); // Закомментировано создание дымовых завес

        this.refreshDimensions();
    }

    private void handleAmmo() {
        boolean hasCreativeAmmo = false;
        for (int i = 0; i < getMaxPassengers(); i++) {
            if (getNthEntity(i) instanceof Player pPlayer && InventoryTool.hasCreativeAmmoBox(pPlayer)) {
                hasCreativeAmmo = true;
            }
        }

        if (hasCreativeAmmo) {
            this.entityData.set(AMMO, 9999);
        } else {
            this.entityData.set(AMMO, countItem(getWeapon(0).ammo));
        }
    }

    @Override
    public boolean canCollideHardBlock() {
        return getDeltaMovement().horizontalDistance() > 0.05 || Mth.abs(this.entityData.get(POWER)) > 0.1;
    }

    @Override
    public boolean canCollideBlockBeastly() {
        return getDeltaMovement().horizontalDistance() > 0.3;
    }

    @Override
    public void vehicleShoot(Player player, int type) {
        boolean hasCreativeAmmo = false;
        for (int i = 0; i < getMaxPassengers() - 1; i++) {
            if (getNthEntity(i) instanceof Player pPlayer && InventoryTool.hasCreativeAmmoBox(pPlayer)) {
                hasCreativeAmmo = true;
            }
        }

        if (this.getWeaponIndex(0) == 0) {
            if (this.cannotFire) {
                return;
            }

            firePosition = (firePosition + 1) % 2; // Переключаем между дулами (0→1→0)

            // Получаем позиции стволов из костей модели
            Vec3 leftBarrelPos, rightBarrelPos;
            if (firePosition == 0) {
                Matrix4f leftBarrelTransform = getLeftBarrel1Position(1.0f);
                Vector4f leftBarrelPosition = transformPosition(leftBarrelTransform, 0, 0, 0);
                leftBarrelPos = new Vec3(leftBarrelPosition.x, leftBarrelPosition.y, leftBarrelPosition.z);
                rightBarrelPos = getRightBarrel1Position(1.0f);
            } else {
                leftBarrelPos = getLeftBarrel2Position(1.0f);
                rightBarrelPos = getRightBarrel2Position(1.0f);
            }

            // Используем общий вектор направления ствола (как в Bmp2Entity)
            Vec3 barrelVector = this.getBarrelVector(1.0F);
            
            // Левый ствол - стреляет всегда
            var leftCannonShell = ((SmallCannonShellWeapon) getWeapon(0)).create(player);
            leftCannonShell.setPos(
                    leftBarrelPos.x - 1.1 * this.getDeltaMovement().x,
                    leftBarrelPos.y,
                    leftBarrelPos.z - 1.1 * this.getDeltaMovement().z
            );
            leftCannonShell.shoot(
                    barrelVector.x,
                    barrelVector.y + 0.005f,
                    barrelVector.z,
                    35.0F,
                    0.25f
            );
            this.level().addFreshEntity(leftCannonShell);
            
            // Правый ствол - стреляет всегда
            var rightCannonShell = ((SmallCannonShellWeapon) getWeapon(0)).create(player);
            rightCannonShell.setPos(
                    rightBarrelPos.x - 1.1 * this.getDeltaMovement().x,
                    rightBarrelPos.y,
                    rightBarrelPos.z - 1.1 * this.getDeltaMovement().z
            );
            rightCannonShell.shoot(
                    barrelVector.x,
                    barrelVector.y + 0.005f,
                    barrelVector.z,
                    35.0F,
                    0.25f
            );
            this.level().addFreshEntity(rightCannonShell);

            // Частицы для левого ствола
            ParticleTool.sendParticle(
                    (ServerLevel)this.level(),
                    ParticleTypes.LARGE_SMOKE,
                    leftBarrelPos.x - 1.1 * this.getDeltaMovement().x,
                    leftBarrelPos.y,
                    leftBarrelPos.z - 1.1 * this.getDeltaMovement().z,
                    1, 0.02, 0.02, 0.02, 0.0F, false
            );
            
            // Частицы для правого ствола - всегда
            ParticleTool.sendParticle(
                    (ServerLevel)this.level(),
                    ParticleTypes.LARGE_SMOKE,
                    rightBarrelPos.x - 1.1 * this.getDeltaMovement().x,
                    rightBarrelPos.y,
                    rightBarrelPos.z - 1.1 * this.getDeltaMovement().z,
                    1, 0.02, 0.02, 0.02, 0.0F, false
            );

            // Увеличиваем счетчик выстрелов
            shotCounter++;
            
            // Звуки от 1-го и 3-го лица воспроизводятся одновременно при каждом выстреле
            if (!player.level().isClientSide) {
                // Звук от 1-го лица
                this.level().playSound(player, this, CowaxSounds.ZENIT_2C6_FIRE_1P.get(), 
                    this.getSoundSource(), 1.0f, 1.0f);
                
                // Звук от 3-го лица с чередованием
                SoundEvent shootSound = (shotCounter % 2 == 0) ? 
                    CowaxSounds.SHOOT_30M.get() : CowaxSounds.SHOOT2_30MM.get();
                
                this.level().playSound(null, this, shootSound, 
                    this.getSoundSource(), 1.0f, 1.0f);
            }

            Level level = player.level();
            Vec3 center = new Vec3(this.getX(), this.getEyeY(), this.getZ());

            for (Entity target : level.getEntitiesOfClass(Entity.class, (new AABB(center, center)).inflate(4.0F)).stream()
                    .sorted(Comparator.comparingDouble((e) -> e.distanceToSqr(center))).toList()) {
                if (target instanceof ServerPlayer serverPlayer) {
                    Mod.PACKET_HANDLER.send(PacketDistributor.PLAYER.with(() -> serverPlayer),
                            new ShakeClientMessage(6.0, 5.0, 9.0, this.getX(), this.getEyeY(), this.getZ()));
                }
            }

            this.entityData.set(CANNON_RECOIL_TIME, 30);
            this.entityData.set(YAW, getTurretYRot());
            this.entityData.set(GUN_FIRE_TIME, Math.min(entityData.get(GUN_FIRE_TIME) + 3, 5));

            this.entityData.set(HEAT, this.entityData.get(HEAT) + 2);
            this.entityData.set(FIRE_ANIM, 3);

            if (hasCreativeAmmo) return;
            // Потребляем 2 боеприпаса (по одному на каждый ствол)
            this.getItemStacks().stream().filter(stack -> stack.is(ModItems.SMALL_SHELL.get())).findFirst().ifPresent(stack -> stack.shrink(1)); // Левый ствол
            this.getItemStacks().stream().filter(stack -> stack.is(ModItems.SMALL_SHELL.get())).findFirst().ifPresent(stack -> stack.shrink(1)); // Правый ствол
        }
    }

    @Override
    public void travel() {
        Entity passenger0 = this.getFirstPassenger();

        if (this.getEnergy() <= 0) return;

        if (!(passenger0 instanceof Player)) {
            this.leftInputDown = false;
            this.rightInputDown = false;
            this.forwardInputDown = false;
            this.backInputDown = false;
            this.entityData.set(POWER, 0f);
        }

        if (forwardInputDown) {
            this.entityData.set(POWER, Math.min(this.entityData.get(POWER) + (this.entityData.get(POWER) < 0 ? 0.001f : 0.0006f) * (1 + getXRot() / 55), 0.12f)); // Замедлили ускорение и максимальную скорость
        }

        if (backInputDown) {
            this.entityData.set(POWER, Math.max(this.entityData.get(POWER) - (this.entityData.get(POWER) > 0 ? 0.001f : 0.0006f) * (1 - getXRot() / 55), -0.015f)); // Замедлили ускорение и максимальную скорость назад
            if (rightInputDown) {
                this.entityData.set(DELTA_ROT, this.entityData.get(DELTA_ROT) + 0.1f);
            } else if (this.leftInputDown) {
                this.entityData.set(DELTA_ROT, this.entityData.get(DELTA_ROT) - 0.1f);
            }
        } else {
            if (rightInputDown) {
                this.entityData.set(DELTA_ROT, this.entityData.get(DELTA_ROT) - 0.1f);
            } else if (this.leftInputDown) {
                this.entityData.set(DELTA_ROT, this.entityData.get(DELTA_ROT) + 0.1f);
            }
        }

        if (this.forwardInputDown || this.backInputDown) {
            this.consumeEnergy(VehicleConfigCowaxPack.ZENIT_2C6_ENERGY_COST.get());
        }

        this.entityData.set(POWER, this.entityData.get(POWER) * (upInputDown ? 0.5f : (rightInputDown || leftInputDown) ? 0.967f : 0.98f));
        this.entityData.set(DELTA_ROT, this.entityData.get(DELTA_ROT) * (float) Math.max(0.76f - 0.1f * this.getDeltaMovement().horizontalDistance(), 0.3));

        double s0 = getDeltaMovement().dot(this.getViewVector(1));

        this.setLeftWheelRot((float) ((this.getLeftWheelRot() - 1.25 * s0) + Mth.clamp(0.75f * this.entityData.get(DELTA_ROT), -5f, 5f)));
        this.setRightWheelRot((float) ((this.getRightWheelRot() - 1.25 * s0) - Mth.clamp(0.75f * this.entityData.get(DELTA_ROT), -5f, 5f)));

        setLeftTrack((float) ((getLeftTrack() - 1.5 * Math.PI * s0) + Mth.clamp(0.4f * Math.PI * this.entityData.get(DELTA_ROT), -5f, 5f)));
        setRightTrack((float) ((getRightTrack() - 1.5 * Math.PI * s0) - Mth.clamp(0.4f * Math.PI * this.entityData.get(DELTA_ROT), -5f, 5f)));

        int i;

        if (entityData.get(L_WHEEL_DAMAGED) && entityData.get(R_WHEEL_DAMAGED)) {
            this.entityData.set(POWER, this.entityData.get(POWER) * 0.93f);
            i = 0;
        } else if (entityData.get(L_WHEEL_DAMAGED)) {
            this.entityData.set(POWER, this.entityData.get(POWER) * 0.975f);
            i = 3;
        } else if (entityData.get(R_WHEEL_DAMAGED)) {
            this.entityData.set(POWER, this.entityData.get(POWER) * 0.975f);
            i = -3;
        } else {
            i = 0;
        }

        if (entityData.get(ENGINE1_DAMAGED)) {
            this.entityData.set(POWER, this.entityData.get(POWER) * 0.85f);
        }

        this.setYRot((float) (this.getYRot() - (isInWater() && !onGround() ? 1.5 : 3.5) * entityData.get(DELTA_ROT) - i * s0));
        if (this.isInWater() || onGround()) {
            this.setDeltaMovement(this.getDeltaMovement().add(getViewVector(1).scale((!isInWater() && !onGround() ? 0.08f : (isInWater() && !onGround() ? 1.2f : 1.5f)) * this.entityData.get(POWER)))); // Замедлили коэффициент движения
        }

        // Consume energy when actually moving
        if ((this.forwardInputDown || this.backInputDown || this.leftInputDown || this.rightInputDown) && this.entityData.get(POWER) > 0.01f) {
            this.consumeEnergy(VehicleConfigCowaxPack.ZENIT_2C6_ENERGY_COST.get());
        }
    }

    @Override
    public void positionRider(@NotNull Entity passenger, @NotNull MoveFunction callback) {
        if (!this.hasPassenger(passenger)) {
            return;
        }

        Matrix4f transform = getTurretTransform(1);
        Matrix4f transformVehicle = getVehicleTransform(1);

        int i = this.getOrderedPassengers().indexOf(passenger);

        var worldPosition = switch (i) {
            case 0 -> transformPosition(transform, 0.0f, 0f, 0f);
            case 1 -> transformPosition(transform, -0.5625f, 0.1f, -0.57275625f);
            case 2 -> transformPosition(transformVehicle, 0.475f, 0.5f, 2.1f);
            case 3 -> transformPosition(transformVehicle, 1.625f, 0.5f, -5.125f);
            case 4 -> transformPosition(transformVehicle, -1.625f, 0.5f, -5.125f);
            default -> throw new IllegalStateException("Unexpected value: " + i);
        };

        passenger.setPos(worldPosition.x, worldPosition.y, worldPosition.z);
        callback.accept(passenger, worldPosition.x, worldPosition.y, worldPosition.z);

        copyEntityData(passenger);
    }

    public Matrix4f getBarrelTransform(float ticks) {
        Matrix4f transformT = getTurretTransform(ticks);

        Matrix4f transform = new Matrix4f();
        // Координаты центра ствола относительно башни (нужно подобрать правильные значения)
        Vector4f worldPosition = transformPosition(transform, 0.01529375f, 3.112525f, -0.1296625f);

        transformT.translate(worldPosition.x, worldPosition.y, worldPosition.z);

        // Наклон применяется ПОСЛЕ вычисления позиции (как в TerminatorEntity)
        float a = getTurretYaw(ticks);

        float r = (Mth.abs(a) - 90f) / 90f;

        float r2;

        if (Mth.abs(a) <= 90f) {
            r2 = a / 90f;
        } else {
            if (a < 0) {
                r2 = -(180f + a) / 90f;
            } else {
                r2 = (180f - a) / 90f;
            }
        }

        float x = Mth.lerp(ticks, turretXRotO, getTurretXRot());
        float xV = Mth.lerp(ticks, xRotO, getXRot());
        float z = Mth.lerp(ticks, prevRoll, getRoll());

        transformT.rotate(Axis.XP.rotationDegrees(x + r * xV + r2 * z));
        return transformT;
    }

    protected void clampRotation(Entity entity) {
        float a = getTurretYaw(1);
        float r = (Mth.abs(a) - 90f) / 90f;

        float r2;

        if (Mth.abs(a) <= 90f) {
            r2 = a / 90f;
        } else {
            if (a < 0) {
                r2 = -(180f + a) / 90f;
            } else {
                r2 = (180f - a) / 90f;
            }
        }

        float min = -74f - r * getXRot() - r2 * getRoll();
        float max = 7.5f - r * getXRot() - r2 * getRoll();

        float f = Mth.wrapDegrees(entity.getXRot());
        float f1 = Mth.clamp(f, min, max);
        entity.xRotO += f1 - f;
        entity.setXRot(entity.getXRot() + f1 - f);

        entity.setYBodyRot(getBarrelYRot(1));
    }

    public @Nullable Vec2 getCameraRotation(float partialTicks, Player player, boolean zoom, boolean isFirstPerson) {

            if (this.getSeatIndex(player) == 0) {

    @OnlyIn(Dist.CLIENT)
    @Override
    public Vec3 getCameraPosition(float partialTicks, Player player, boolean zoom, boolean isFirstPerson) {
        if (zoom || isFirstPerson) {
            if (this.getSeatIndex(player) == 0) {
                if (zoom) {
                    return new Vec3(this.driverZoomPos(partialTicks).x, Mth.lerp(partialTicks, player.yo + player.getEyeHeight(), player.getEyeY()), this.driverZoomPos(partialTicks).z);
                } else {
                    // Обычная камера - поднимаем и сдвигаем вперед
                    return new Vec3(Mth.lerp(partialTicks, player.xo, player.getX()), 
                                  Mth.lerp(partialTicks, player.yo + player.getEyeHeight(), player.getEyeY()), 
                                  Mth.lerp(partialTicks, player.zo, player.getZ()));
                }
            } else if (this.getSeatIndex(player) == 1) {
                return new Vec3(Mth.lerp(partialTicks, player.xo, player.getX()), Mth.lerp(partialTicks, player.yo + player.getEyeHeight(), player.getEyeY()), Mth.lerp(partialTicks, player.zo, player.getZ()));
            }
        }
        return super.getCameraPosition(partialTicks, player, false, false);
    }
